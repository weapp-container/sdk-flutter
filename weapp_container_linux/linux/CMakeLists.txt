cmake_minimum_required(VERSION 3.13)
project(weapp_container_linux LANGUAGES CXX)
set(PLUGIN_NAME "weapp_container_linux_plugin")
set(NATIVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor")
if(NOT EXISTS "${NATIVE_ROOT}/lib/libweapp_container_sdk.a")
  message(FATAL_ERROR "Prepare the local SDK first: node native_desktop/prepare.mjs --target linux --from <installed-desktop-sdk>")
endif()
add_library(${PLUGIN_NAME} SHARED weapp_container_linux_plugin.cc native_shared/src/registry.cpp)
apply_standard_settings(${PLUGIN_NAME})
set_target_properties(${PLUGIN_NAME} PROPERTIES CXX_VISIBILITY_PRESET hidden)
target_compile_features(${PLUGIN_NAME} PRIVATE cxx_std_17)
target_include_directories(${PLUGIN_NAME} INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_include_directories(${PLUGIN_NAME} PRIVATE native_shared/include native_shared/third_party "${NATIVE_ROOT}/include")
target_link_libraries(${PLUGIN_NAME} PRIVATE flutter PkgConfig::GTK "${NATIVE_ROOT}/lib/libweapp_container_sdk.a" pthread dl m fontconfig freetype)
set(weapp_container_linux_bundled_libraries "" PARENT_SCOPE)
