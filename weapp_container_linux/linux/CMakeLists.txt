cmake_minimum_required(VERSION 3.13)
project(weapp_container_linux LANGUAGES CXX)
set(PLUGIN_NAME "weapp_container_linux_plugin")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  message(FATAL_ERROR "The public preview supports x86_64 on this platform")
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/runtime.cmake")
  message(FATAL_ERROR "Use the staged pub.dev package; tools/package-public.mjs generates the reviewed runtime pin")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/runtime.cmake")
if(NOT EXISTS "${NATIVE_ROOT}/lib/libweapp_container_sdk.a")
  message(FATAL_ERROR "Prepare the local SDK first: node native_desktop/prepare.mjs --target linux --from <installed-desktop-sdk>")
endif()
if(NOT EXISTS "${NATIVE_ROOT}/lib/libweapp_container_desktop.a" OR NOT EXISTS "${NATIVE_ROOT}/include/weapp/native_preview.hpp" OR NOT EXISTS "${NATIVE_ROOT}/include/weapp/device_apis.hpp")
  message(FATAL_ERROR "The reviewed desktop native device/preview wrapper is required; re-run prepare with the matching SDK version")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(EPOXY REQUIRED IMPORTED_TARGET epoxy)
pkg_check_modules(DEVICE_GIO REQUIRED IMPORTED_TARGET gio-2.0>=2.66)
pkg_check_modules(CURL QUIET IMPORTED_TARGET libcurl>=7.85)
add_library(${PLUGIN_NAME} SHARED weapp_container_linux_plugin.cc native_shared/src/registry.cpp)
apply_standard_settings(${PLUGIN_NAME})
set_target_properties(${PLUGIN_NAME} PROPERTIES CXX_VISIBILITY_PRESET hidden)
target_compile_features(${PLUGIN_NAME} PRIVATE cxx_std_17)
target_include_directories(${PLUGIN_NAME} INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_include_directories(${PLUGIN_NAME} PRIVATE native_shared/include native_shared/third_party "${NATIVE_ROOT}/include")
target_link_libraries(${PLUGIN_NAME} PRIVATE flutter PkgConfig::GTK PkgConfig::EPOXY PkgConfig::DEVICE_GIO "${NATIVE_ROOT}/lib/libweapp_container_desktop.a" "${NATIVE_ROOT}/lib/libweapp_container_sdk.a" pthread dl m fontconfig freetype)
if(TARGET PkgConfig::CURL)
  target_link_libraries(${PLUGIN_NAME} PRIVATE PkgConfig::CURL)
endif()
set(weapp_container_linux_bundled_libraries "" PARENT_SCOPE)
